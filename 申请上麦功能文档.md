## 一、前端调用方式

### 1.1 进入语音房间

```typescript
// 文件: app/(tabs)/living.tsx

// 作为观众加入语音房间
const joinVoiceRoom = async (roomId: string) => {
  const userInfo = await getUserInfo();
  
  // 1. 获取 UserSig
  const resp = await fetch(`${API_BASE_URL}/live/join`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      userId: userInfo.unique_id,
      roomId: roomId,
    }),
  });
  
  const data = await resp.json();
  
  // 2. 登录 TUILiveKit
  await TUILiveModule.login(data.data.sdkAppId, data.data.userId, data.data.userSig);
  
  // 3. 设置用户信息
  await TUILiveModule.setSelfInfo(userInfo.nickname, userInfo.avatar);
  
  // 4. 进入语音房间
  await TUILiveModule.joinVoiceRoom({
    roomId: roomId,
    nickname: userInfo.nickname,
    avatar: userInfo.avatar
  });
};
```

### 1.2 TUILiveModule 原生模块接口

| 方法 | 参数 | 说明 |
|------|------|------|
| `login(sdkAppId, userId, userSig)` | SDK配置 | 登录腾讯云 |
| `setSelfInfo(nickname, avatar)` | 用户信息 | 设置头像昵称 |
| `startVoiceRoom(options)` | `{roomId, roomName, nickname, avatar}` | 创建语聊房(主播) |
| `joinVoiceRoom(options)` | `{roomId, nickname, avatar}` | 加入语聊房(观众) |

## 二、申请上麦流程

### 2.1 用户操作流程

```
观众进入房间 → 点击上麦按钮 → 发送上麦申请 → 等待房主审批
                                              │
                    ┌─────────────────────────┴─────────────────────────┐
                    ▼                                                   ▼
               房主同意                                              房主拒绝
                    │                                                   │
                    ▼                                                   ▼
              上麦成功，开启麦克风                                   显示拒绝提示
```

### 2.2 上麦状态

```kotlin
// 文件: voiceroom/state/SeatState.kt

enum class LinkStatus {
    NONE,      // 默认状态，未上麦
    APPLYING,  // 申请中，等待房主审批
    LINKING    // 已上麦，正在连麦
}
```

### 2.3 核心代码实现

#### 观众端 - 申请上麦

```kotlin
// 文件: voiceroom/view/bottommenu/AudienceFunctionView.kt

private fun takeSeat() {
    if (mSeatState.linkStatus.value == SeatState.LinkStatus.APPLYING) return

    // 调用 SeatGridView.takeSeat() 发起上麦申请
    // 参数: index=-1 表示自动分配座位, timeout=60 表示60秒超时
    mSeatGridView.takeSeat(-1, 60, object : VoiceRoomDefine.RequestCallback {
        
        override fun onAccepted(userInfo: TUIRoomDefine.UserInfo) {
            // 房主同意，上麦成功
            mSeatManager.updateLinkState(SeatState.LinkStatus.LINKING)
        }

        override fun onRejected(userInfo: TUIRoomDefine.UserInfo) {
            // 房主拒绝
            mSeatManager.updateLinkState(SeatState.LinkStatus.NONE)
            ToastUtil.toastShortMessage("上麦申请被拒绝")
        }

        override fun onCancelled(userInfo: TUIRoomDefine.UserInfo) {
            // 用户取消申请
            mSeatManager.updateLinkState(SeatState.LinkStatus.NONE)
        }

        override fun onTimeout(userInfo: TUIRoomDefine.UserInfo) {
            // 申请超时
            mSeatManager.updateLinkState(SeatState.LinkStatus.NONE)
            ToastUtil.toastShortMessage("上麦申请超时")
        }

        override fun onError(userInfo: TUIRoomDefine.UserInfo, error: TUICommonDefine.Error, message: String) {
            // 申请失败
            mSeatManager.updateLinkState(SeatState.LinkStatus.NONE)
            ErrorLocalized.onError(error)
        }
    })
    
    // 更新状态为申请中
    mSeatManager.updateLinkState(SeatState.LinkStatus.APPLYING)
}
```

#### 取消上麦申请

```kotlin
private fun cancelSeatApplication(view: View) {
    view.isEnabled = false
    mSeatGridView.cancelRequest("", TUIActionCallback(
        success = {
            mSeatManager.updateLinkState(SeatState.LinkStatus.NONE)
            view.isEnabled = true
        },
        error = { errorCode, message ->
            ErrorLocalized.onError(errorCode)
            view.isEnabled = true
        }
    ))
}
```

#### 下麦

```kotlin
private fun leaveSeat() {
    mSeatGridView.leaveSeat(TUIActionCallback(
        success = {
            // 下麦成功
        },
        error = { errorCode, message ->
            if (errorCode != TUICommonDefine.Error.SUCCESS) {
                ErrorLocalized.onError(errorCode)
            }
        }
    ))
}
```

### 2.4 SeatGridView 核心方法

```kotlin
// 文件: voiceroomcore/SeatGridView.kt

/**
 * 申请上麦
 * @param index 座位索引，-1表示自动分配
 * @param timeout 超时时间(秒)
 * @param callback 回调
 */
fun takeSeat(index: Int, timeout: Int, callback: VoiceRoomDefine.RequestCallback?) {
    val liveInfo = liveListStore.liveState.currentLive.value
    val isOwner = TUIRoomEngine.getSelfInfo().userId == liveInfo.liveOwner.userID
    
    // 自由上麦模式 或 房主：直接上麦
    if (liveInfo.seatMode == TakeSeatMode.FREE || isOwner) {
        seatStore?.takeSeat(index, object : CompletionHandler {
            override fun onSuccess() {
                callback?.onAccepted(userInfo)
            }
            override fun onFailure(code: Int, desc: String) {
                callback?.onError(userInfo, TUICommonDefine.Error.fromInt(code), desc)
            }
        })
        return
    }

    // 需要审批模式：发送上麦申请
    coGuestStore?.applyForSeat(index, timeout, null, object : CompletionHandler {
        override fun onSuccess() { }
        override fun onFailure(code: Int, desc: String) {
            callback?.onError(convertToUserInfo(liveInfo.liveOwner), TUICommonDefine.Error.fromInt(code), desc)
        }
    })
    applicationCallback = callback
}

/**
 * 下麦
 */
fun leaveSeat(callback: TUIRoomDefine.ActionCallback?) {
    val liveInfo = liveListStore.liveState.currentLive.value
    if (liveInfo.seatMode == TakeSeatMode.FREE || isOwner()) {
        seatStore?.leaveSeat(ActionCallback(callback))
    } else {
        coGuestStore?.disconnect(ActionCallback(callback))
    }
}

/**
 * 取消上麦申请
 */
fun cancelRequest(requestId: String, callback: TUIRoomDefine.ActionCallback?) {
    coGuestStore?.cancelSeatApplication(ActionCallback(callback))
}
```

## 三、上麦模式

### 3.1 两种模式

| 模式 | 说明 | 行为 |
|------|------|------|
| `FREE` | 自由上麦 | 观众可直接上麦，无需审批 |
| `APPLY_TO_TAKE` | 申请上麦 | 观众需发送申请，房主同意后才能上麦 |

### 3.2 房主设置上麦模式

```kotlin
// 房主可以在连麦管理面板中设置
fun updateSeatMode(seatMode: TUIRoomDefine.SeatMode, callback: TUIRoomDefine.ActionCallback?) {
    seatStore?.updateSeatMode(seatMode, ActionCallback(callback))
}
```

## 四、常见错误码

| 错误码 | 常量 | 说明 |
|--------|------|------|
| 100200 | `MIC_SEAT_IS_LOCKED` | 麦位已被锁定 |
| 100201 | `SEAT_IS_ALREADY_OCCUPIED` | 座位已被占用 |
| 100202 | `ALREADY_ON_THE_MIC_QUEUE` | 已在上麦队列中 |
| 100203 | `ALREADY_ON_THE_MIC` | 已经在麦上 |
| 100204 | `NOT_ON_THE_MIC_QUEUE` | 不在上麦队列中 |
| 100205 | `THE_SEATS_ARE_ALL_TAKEN` | 所有座位已满 |
| 100206 | `NOT_ON_THE_MIC_SEAT` | 不在麦位上 |
